64. Validacija hierarhije
function validateAccountHierarchy(
  accountCode: string,
  parentCode: string | null
) {
  if (!parentCode) return true;

  if (accountCode === parentCode) {
    throw new Error("Konto ne more biti sam sebi nadrejeni konto.");
  }

  if (!accountCode.startsWith(parentCode)) {
    throw new Error(
      "Konto mora biti hierarhično povezan z nadrejenim kontom."
    );
  }

  return true;
}

function validateJournalEntry(
  entry: JournalEntryInput
) {
  if (entry.lines.length < 2) {
    throw new Error(
      "Knjižba mora imeti najmanj dve vrstici."
    );
  }

  const totalDebit = entry.lines.reduce(
    (sum, line) => sum + line.debit,
    0
  );

  const totalCredit = entry.lines.reduce(
    (sum, line) => sum + line.credit,
    0
  );

  if (totalDebit !== totalCredit) {
    throw new Error(
      "Knjižba ni uravnotežena."
    );
  }
}

96. Neizpodbitna računovodska invarianta

Sistem mora zagotavljati:

Σ DEBET = Σ KREDIT

Toda v produkciji moramo preverjati tudi:

Σ DEBET - Σ KREDIT = 0

DEBET >= 0
KREDIT >= 0

DEBET > 0
IN
KREDIT > 0

async function postJournalEntry(input) {
  return db.transaction(async (tx) => {

    await validateFiscalPeriod(input);

    await validateAccounts(input.lines);

    await validateDimensions(input.lines);

    validateBalancedEntry(input.lines);

    const entry =
      await createJournalEntry(tx, input);

    await createJournalLines(
      tx,
      entry.id,
      input.lines
    );

    await createSubledgerItems(
      tx,
      entry.id
    );

    await createTaxRecords(
      tx,
      entry.id
    );

    await createAuditLog(
      tx,
      entry.id
    );

    return entry;
  });
}
